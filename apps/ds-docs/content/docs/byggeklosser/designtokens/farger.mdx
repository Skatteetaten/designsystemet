import paletteJson from '@skatteetaten/ds-core-designtokens/designtokens/palette.json';

import { TokenCardList } from '../../../../app/components/token-palette';

<TokenCardList
  id={'base-colors'}
  title={'Basefarger'}
  tokens={Object.fromEntries(
    Object.entries(paletteJson[':root,\n:host']).filter(
      ([name]) => name.includes('palette') && !name.includes('alpha')
    )
  )}
/>

<TokenCardList
  id={'semantic-colors'}
  title={'Sidestyling'}
  description={'Bakgrunnsfarger, rammer og tekstfarger du ofte bruker.'}
  showValue={false}
  icon={'sidestyling'}
  tokens={Object.fromEntries(
    Object.entries(paletteJson[':root,\n:host']).filter(
      ([name]) =>
        name.includes('semantic-page') || name.includes('semantic-divider')
    )
  )}
/>

<TokenCardList
  title={'Interaksjon'}
  description={'Farger som brukes til elementer du kan klikke på.'}
  showValue={false}
  icon={'interaksjon'}
  tokens={Object.fromEntries(
    Object.entries(paletteJson[':root,\n:host']).filter(([name]) =>
      name.includes('interactive')
    )
  )}
/>

<TokenCardList
  title={'Statusfarger'}
  description={
    'Farger som brukes til positive, nøytrale eller negative statusmeldinger.'
  }
  showValue={false}
  icon={'status'}
  tokens={Object.fromEntries(
    Object.entries(paletteJson[':root,\n:host']).filter(
      ([name]) =>
        name.includes('success') ||
        name.includes('warning') ||
        name.includes('danger') ||
        name.includes('disabled')
    )
  )}
/>
