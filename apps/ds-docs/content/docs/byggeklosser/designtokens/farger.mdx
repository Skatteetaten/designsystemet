<Table caption={'Fargepalett'} size={'medium'}>
  <Table.Header>
    <Table.Row>
      <Table.HeaderCell scope={'col'}>{'Farge'}</Table.HeaderCell>

  </Table.Header>
  <Table.Body>
    {Object.entries(paletteJson[':root,\n:host']).map(([name, value]) => (
      <Table.Row key={name}>

          <span
            aria-label={`${name} farge`}
            style={{
              display: 'inline-block',
              width: '1.5rem',
              height: '1.5rem',
              backgroundColor: value,
              verticalAlign: 'middle',
            }}
          />
        </Table.DataCell>
        <Table.DataCell>{name}</Table.DataCell>
        <Table.DataCell>
          <IconButton
            svgPath={CopySVGpath}
            title={`Kopier ${name}`}

            size={'small'}
          />
        </Table.DataCell>
      </Table.Row>
    ))}
  </Table.Body>
